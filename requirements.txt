# ---------------------------------------------------------------------------
#  Pinned to the stack that produced the released checkpoints.
#
#  transformers is the one that really matters.  The checkpoints' config.json
#  records 4.46.1, and `from_pretrained` does NOT round-trip these configs on
#  transformers 5.x -- it raises rather than loading.  Either install the pin
#  below, or load the weights manually (see README "Loading a checkpoint"):
#
#      from safetensors.torch import load_file
#      cfg = SLTConfig.from_pretrained(ckpt)
#      model = EEG2Source2EEG_autoregressive(cfg)
#      model.load_state_dict(load_file(ckpt + '/model.safetensors'), strict=False)
#
#  `strict=False` is required: the sinusoidal positional-encoding tables are
#  deterministic buffers that are rebuilt identically at __init__.
# ---------------------------------------------------------------------------
